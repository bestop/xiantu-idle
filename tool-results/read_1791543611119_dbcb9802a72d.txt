    58→    58→interface UseGameStateReturn {
    59→    59→  // 状态
    60→    60→  character: Character | null;
    61→    61→  inventory: InventoryItem[];
    62→    62→  battle: BattleState;
    63→    63→  logs: GameLogEntry[];
    64→    64→  market: MarketListing[];
    65→    65→  currentTab: GameTab;
    66→    66→  isLoading: boolean;
    67→    67→  playerId: string;
    68→    68→  
    69→    69→  // 操作

    71→    71→  resetGame: () => void;
    72→    72→  setTab: (tab: GameTab) => void;
    73→    73→  addLog: (type: GameLogEntry['type'], message: string) => void;
    74→    74→  setCharacter: (character: Character | ((prev: Character | null) => Character | null)) => void;
    75→    75→  setInventory: (inventory: InventoryItem[] | ((prev: InventoryItem[]) => InventoryItem[])) => void;
    76→    76→

    79→    79→  quickBattle: (isGodMode?: boolean, monster?: Monster) => void;
    80→    80→  endBattle: () => void;
    81→    81→  mapEncounter: (monster: Monster, isGodMode?: boolean) => void;
    82→    82→  
    83→    83→  // 物品相关
    84→    84→  useItem: (item: GameItem, quantity?: number) => void;
    85→    85→  equip: (item: GameItem) => void;
    86→    86→  unequip: (slot: 'weapon' | 'armor' | 'accessory') => void;
    87→    87→  sellItem: (item: GameItem, price: number) => void;
    88→    88→  buyFromMarket: (listingId: string) => void;
    89→    89→  buyNpcItem: (item: GameItem, price: number) => void;
    90→    90→  craftItem: (recipe: CraftingRecipe) => void;
    91→    91→  
    92→    92→  // 其他操作
    93→    93→  doMeditate: () => void;
    94→    94→  doTribulation: () => { success: boolean; message: string };
    95→    95→  restore: () => void;
    96→    96→  
    97→    97→  // 宠物相关
    98→    98→  buyPet: (petId: string, price: number) => void;
    99→    99→  activatePet: (petId: string) => void;
   100→   100→  deactivatePet: (petId: string) => void;
   101→   101→  renamePet: (petId: string, nickname: string) => void;
   102→   102→  evolvePet: (petId: string) => void;
   103→   103→  levelUpPet: (petId: string, exp: number) => void;
   104→   104→  
   105→   105→  // 技能系统
   106→   106→  unlockSkill: (skillId: string) => void;
   107→   107→  useSkill: (skillId: string) => void;
   108→   108→  upgradeSkill: (skillId: string, levels?: number) => void;
   109→   109→  
   110→   110→  // 轮回系统
   111→   111→  doSamsara: () => void;
   112→   112→}

   114→   114→// 惰性初始化函数
   115→   115→function getInitialState() {
   116→   116→  return {
   117→   117→    character: loadCharacter(),
   118→   118→    inventory: loadInventory(),
   119→   119→    logs: loadLogs(),
   120→   120→    market: loadMarket(),
   121→   121→    playerId: getPlayerId()
   122→   122→  };
   123→   123→}
   124→   124→
   125→   125→export function useGameState(): UseGameStateReturn {
   126→   126→  const [character, setCharacter] = useState<Character | null>(null);
   127→   127→  const [inventory, setInventory] = useState<InventoryItem[]>([]);
   128→   128→  const [battle, setBattle] = useState<BattleState>({
   129→   129→    inBattle: false,
   130→   130→    monster: null,
   131→   131→    playerHp: 0,
   132→   132→    monsterHp: 0,
   133→   133→    battleLog: [],
   134→   134→    isAuto: false,
   135→   135→    result: null
   136→   136→  });
   137→   137→  const [logs, setLogs] = useState<GameLogEntry[]>([]);
   138→   138→  const [market, setMarket] = useState<MarketListing[]>([]);
   139→   139→  const [currentTab, setCurrentTab] = useState<GameTab>('battle');
   140→   140→  const [isLoading, setIsLoading] = useState(true);
   141→   141→  const [playerId, setPlayerId] = useState<string>('');
   142→   142→  const [isInitialized, setIsInitialized] = useState(false);
   143→   143→
   144→   144→  // 客户端挂载后初始化数据 - 使用回调形式避免同步setState警告
   145→   145→  useEffect(() => {
   146→   146→    // 使用函数式更新，React会正确处理
   147→   147→    const loadGameData = () => {
   148→   148→      const initialState = getInitialState();
   149→   149→      setCharacter(initialState.character);
   150→   150→      setInventory(initialState.inventory);
   151→   151→      setLogs(initialState.logs);
   152→   152→      setMarket(initialState.market);
   153→   153→      setPlayerId(initialState.playerId);
   154→   154→      setIsLoading(false);
   155→   155→      setIsInitialized(true);
   156→   156→    };
   157→   157→    
   158→   158→    // 使用 setTimeout 0 确保在渲染完成后执行
   159→   159→    const timer = setTimeout(loadGameData, 0);
   160→   160→    return () => clearTimeout(timer);
   161→   161→  }, []);
   162→   162→
   163→   163→  // 自动保存 - 只在初始化完成后保存
   164→   164→  useEffect(() => {
   165→   165→    if (isInitialized && character) {
   166→   166→      saveCharacter(character);
   167→   167→    }
   168→   168→  }, [character, isInitialized]);
   169→   169→
   170→   170→  useEffect(() => {
   171→   171→    if (isInitialized) {
   172→   172→      saveInventory(inventory);
   173→   173→    }
   174→   174→  }, [inventory, isInitialized]);
   175→   175→
   176→   176→  useEffect(() => {
   177→   177→    if (isInitialized) {
   178→   178→      saveLogs(logs);
   179→   179→    }
   180→   180→  }, [logs, isInitialized]);
   181→   181→
   182→   182→  useEffect(() => {
   183→   183→    if (isInitialized) {
   184→   184→      saveMarket(market);
   185→   185→    }
   186→   186→  }, [market, isInitialized]);
   187→   187→
   188→   188→  // 添加日志
   189→   189→  const addLog = useCallback((type: GameLogEntry['type'], message: string) => {
   190→   190→    const newLog: GameLogEntry = {
   191→   191→      id: generateId(),
   192→   192→      timestamp: Date.now(),
   193→   193→      type,
   194→   194→      message
   195→   195→    };
   196→   196→    setLogs(prev => [...prev.slice(-99), newLog]);
   197→   197→  }, []);
   198→   198→
   199→   199→  // 初始化游戏
   200→   200→  const initGame = useCallback((name: string) => {
   201→   201→    const newCharacter = createNewCharacter(name);
   202→   202→    setCharacter(newCharacter);
   203→   203→    setInventory([]);
   204→   204→    setLogs([]);
   205→   205→    addLog('system', `${name}踏上了修仙之路...`);
   206→   206→  }, [addLog]);
   207→   207→
   208→   208→  // 重置游戏
   209→   209→  const resetGame = useCallback(() => {
   210→   210→    clearAllData();
   211→   211→    setCharacter(null);
   212→   212→    setInventory([]);
   213→   213→    setLogs([]);
   214→   214→    setMarket([]);
   215→   215→    setBattle({
   216→   216→      inBattle: false,
   217→   217→      monster: null,
   218→   218→      playerHp: 0,
   219→   219→      monsterHp: 0,
   220→   220→      battleLog: [],
   221→   221→      isAuto: false,
   222→   222→      result: null
   223→   223→    });
   224→   224→    setPlayerId(getPlayerId());
   225→   225→  }, []);
   226→   226→
   227→   227→  // 设置标签页
   228→   228→  const setTab = useCallback((tab: GameTab) => {
   229→   229→    setCurrentTab(tab);
   230→   230→  }, []);
   231→   231→
   232→   232→  // 开始战斗
   233→   233→  const startBattle = useCallback((monster: Monster) => {
   234→   234→    if (!character) return;
   235→   235→    
   236→   236→    setBattle({
   237→   237→      inBattle: true,
   238→   238→      monster,
   239→   239→      playerHp: character.stats.hp,
   240→   240→      monsterHp: monster.hp,
   241→   241→      battleLog: [],
   242→   242→      isAuto: false,
   243→   243→      result: null
   244→   244→    });
   245→   245→    
   246→   246→    addLog('battle', `开始与${monster.name}战斗！`);
   247→   247→  }, [character, addLog]);
   248→   248→
   249→   249→  // 快速战斗
   250→   250→  const quickBattle = useCallback((isGodMode: boolean = false, monster?: Monster) => {
   251→   251→    if (!character || battle.inBattle) return;
   252→   252→    
   253→   253→    const targetMonster = monster || getRecommendedMonster(character.realm, character.level);
   254→   254→    if (!targetMonster) return;
   255→   255→    
   256→   256→    // 执行战斗
   257→   257→    const { logs: battleLogs, result, finalHp } = executeBattle(character, targetMonster, isGodMode);
   258→   258→    
   259→   259→    // 更新角色状态
   260→   260→    let updatedCharacter = { ...character };
   261→   261→    updatedCharacter.stats.hp = finalHp;
   262→   262→    
   263→   263→    // 处理战斗结果
   264→   264→    if (result === 'win') {
   265→   265→      // 获得经验和金币
   266→   266→      const expResult = addExperience(updatedCharacter, targetMonster.exp);
   267→   267→      updatedCharacter = expResult.character;
   268→   268→      updatedCharacter.gold += targetMonster.gold;
   269→   269→      
   270→   270→      // 获得掉落物
   271→   271→      const drops = calculateDrops(targetMonster);
   272→   272→      let newInventory = [...inventory];
   273→   273→      drops.forEach(item => {
   274→   274→        newInventory = addToInventory(newInventory, item, 1);
   275→   275→        addLog('item', `获得了${item.name}！`);
   276→   276→      });
   277→   277→      setInventory(newInventory);
   278→   278→      
   279→   279→      addLog('battle', `击败了${targetMonster.name}，获得${targetMonster.exp}经验和${targetMonster.gold}金币`);
   280→   280→      
   281→   281→      if (expResult.leveledUp) {
   282→   282→        addLog('level_up', `恭喜！升级到了${updatedCharacter.realm}${expResult.newLevel}层！`);
   283→   283→      }
   284→   284→      
   285→   285→      // 更新宠物战斗次数、经验和忠诚度
   286→   286→      if (updatedCharacter.pets && updatedCharacter.pets.length > 0) {
   287→   287→        updatedCharacter.pets = updatedCharacter.pets.map(pet => {
   288→   288→          if (pet.isActive) {
   289→   289→            const petExpGain = Math.floor(targetMonster.exp * 0.3);
   290→   290→            const loyaltyGain = 1;
   291→   291→            const updatedPet = { ...pet.pet };
   292→   292→            updatedPet.exp += petExpGain;
   293→   293→            updatedPet.loyalty = Math.min(100, updatedPet.loyalty + loyaltyGain);
   294→   294→            
   295→   295→            // 检查宠物升级
   296→   296→            while (updatedPet.exp >= updatedPet.maxExp) {
   297→   297→              const leveledPet = levelUpPetUtil(updatedPet);
   298→   298→              updatedPet.level = leveledPet.level;
   299→   299→              updatedPet.exp = leveledPet.exp;
   300→   300→              updatedPet.maxExp = leveledPet.maxExp;
   301→   301→              updatedPet.stats = leveledPet.stats;
   302→   302→              addLog('pet', `宠物 ${updatedPet.name} 升级了！`);
   303→   303→            }
   304→   304→            
   305→   305→            return {
   306→   306→              ...pet,
   307→   307→              pet: updatedPet,
   308→   308→              battleCount: (pet.battleCount || 0) + 1,
   309→   309→              winCount: (pet.winCount || 0) + 1
   310→   310→            };
   311→   311→          }
   312→   312→          return pet;
   313→   313→        });
   314→   314→      }
   315→   315→    } else {
   316→   316→      addLog('battle', `被${targetMonster.name}击败了...`);
   317→   317→      
   318→   318→      // 更新宠物战斗次数（失败也计入）
   319→   319→      if (updatedCharacter.pets && updatedCharacter.pets.length > 0) {
   320→   320→        updatedCharacter.pets = updatedCharacter.pets.map(pet => {
   321→   321→          if (pet.isActive) {
   322→   322→            return {
   323→   323→              ...pet,
   324→   324→              battleCount: (pet.battleCount || 0) + 1
   325→   325→            };
   326→   326→          }
   327→   327→          return pet;
   328→   328→        });
   329→   329→      }
   330→   330→    }
   331→   331→    
   332→   332→    // 更新血量
   333→   333→    const stats = calculateStatsWithEquipment(updatedCharacter);
   334→   334→    updatedCharacter.stats = { ...updatedCharacter.stats, maxHp: stats.maxHp, maxMp: stats.maxMp };
   335→   335→    
   336→   336→    setCharacter(updatedCharacter);
   337→   337→    setBattle({
   338→   338→      inBattle: false,
   339→   339→      monster: targetMonster,
   340→   340→      playerHp: finalHp,
   341→   341→      monsterHp: 0,
   342→   342→      battleLog: battleLogs,
   343→   343→      isAuto: false,
   344→   344→      result
   345→   345→    });
   346→   346→  }, [character, battle.inBattle, inventory, addLog]);
   347→   347→
   348→   348→  // 地图遇怪战斗
   349→   349→  const mapEncounter = useCallback((monster: Monster, isGodMode: boolean = false) => {
   350→   350→    if (!character) return;
   351→   351→    
   352→   352→    // 执行战斗
   353→   353→    const { logs: battleLogs, result, finalHp } = executeBattle(character, monster, isGodMode);
   354→   354→    
   355→   355→    // 更新角色状态
   356→   356→    let updatedCharacter = { ...character };
   357→   357→    updatedCharacter.stats.hp = finalHp;
   358→   358→    
   359→   359→    // 处理战斗结果
   360→   360→    if (result === 'win') {
   361→   361→      // 获得经验和金币（地图怪物奖励+20%）
   362→   362→      const bonusExp = Math.floor(monster.exp * 1.2);
   363→   363→      const bonusGold = Math.floor(monster.gold * 1.2);
   364→   364→      
   365→   365→      const expResult = addExperience(updatedCharacter, bonusExp);
   366→   366→      updatedCharacter = expResult.character;
   367→   367→      updatedCharacter.gold += bonusGold;
   368→   368→      
   369→   369→      // 获得掉落物
   370→   370→      const drops = calculateDrops(monster);
   371→   371→      let newInventory = [...inventory];
   372→   372→      drops.forEach(item => {
   373→   373→        newInventory = addToInventory(newInventory, item, 1);
   374→   374→        addLog('item', `获得了${item.name}！`);
   375→   375→      });
   376→   376→      setInventory(newInventory);
   377→   377→      
   378→   378→      addLog('battle', `击败了${monster.name}，获得${bonusExp}经验和${bonusGold}金币（地图奖励+20%）`);
   379→   379→      
   380→   380→      if (expResult.leveledUp) {
   381→   381→        addLog('level_up', `恭喜！升级到了${updatedCharacter.realm}${expResult.newLevel}层！`);
   382→   382→      }
   383→   383→      
   384→   384→      // 更新宠物战斗次数、经验和忠诚度
   385→   385→      if (updatedCharacter.pets && updatedCharacter.pets.length > 0) {
   386→   386→        const bonusExp = Math.floor(monster.exp * 1.2);
   387→   387→        updatedCharacter.pets = updatedCharacter.pets.map(pet => {
   388→   388→          if (pet.isActive) {
   389→   389→            const petExpGain = Math.floor(bonusExp * 0.3);
   390→   390→            const loyaltyGain = 1;
   391→   391→            const updatedPet = { ...pet.pet };
   392→   392→            updatedPet.exp += petExpGain;
   393→   393→            updatedPet.loyalty = Math.min(100, updatedPet.loyalty + loyaltyGain);
   394→   394→            
   395→   395→            // 检查宠物升级
   396→   396→            while (updatedPet.exp >= updatedPet.maxExp) {
   397→   397→              const leveledPet = levelUpPetUtil(updatedPet);
   398→   398→              updatedPet.level = leveledPet.level;
   399→   399→              updatedPet.exp = leveledPet.exp;
   400→   400→              updatedPet.maxExp = leveledPet.maxExp;
   401→   401→              updatedPet.stats = leveledPet.stats;
   402→   402→              addLog('pet', `宠物 ${updatedPet.name} 升级了！`);
   403→   403→            }
   404→   404→            
   405→   405→            return {
   406→   406→              ...pet,
   407→   407→              pet: updatedPet,
   408→   408→              battleCount: (pet.battleCount || 0) + 1,
   409→   409→              winCount: (pet.winCount || 0) + 1
   410→   410→            };
   411→   411→          }
   412→   412→          return pet;
   413→   413→        });
   414→   414→      }
   415→   415→    } else {
   416→   416→      addLog('battle', `被${monster.name}击败了...`);
   417→   417→      
   418→   418→      // 更新宠物战斗次数（失败也计入）
   419→   419→      if (updatedCharacter.pets && updatedCharacter.pets.length > 0) {
   420→   420→        updatedCharacter.pets = updatedCharacter.pets.map(pet => {
   421→   421→          if (pet.isActive) {
   422→   422→            return {
   423→   423→              ...pet,
   424→   424→              battleCount: (pet.battleCount || 0) + 1
   425→   425→            };
   426→   426→          }
   427→   427→          return pet;
   428→   428→        });
   429→   429→      }
   430→   430→    }
   431→   431→    
   432→   432→    // 更新血量
   433→   433→    const stats = calculateStatsWithEquipment(updatedCharacter);
   434→   434→    updatedCharacter.stats = { ...updatedCharacter.stats, maxHp: stats.maxHp, maxMp: stats.maxMp };
   435→   435→    
   436→   436→    setCharacter(updatedCharacter);
   437→   437→    setBattle({
   438→   438→      inBattle: false,
   439→   439→      monster,
   440→   440→      playerHp: finalHp,
   441→   441→      monsterHp: 0,
   442→   442→      battleLog: battleLogs,
   443→   443→      isAuto: false,
   444→   444→      result
   445→   445→    });
   446→   446→  }, [character, inventory, addLog]);
   447→   447→
   448→   448→  // 从NPC商店购买物品
   449→   449→  const buyNpcItem = useCallback((item: GameItem, price: number) => {
   450→   450→    if (!character || character.gold < price) return;
   451→   451→    
   452→   452→    // 扣除金币
   453→   453→    const updatedCharacter = {
   454→   454→      ...character,
   455→   455→      gold: character.gold - price
   456→   456→    };
   457→   457→    
   458→   458→    // 添加物品到背包
   459→   459→    const newInventory = addToInventory(inventory, item, 1);
   460→   460→    
   461→   461→    setCharacter(updatedCharacter);
   462→   462→    setInventory(newInventory);
   463→   463→    addLog('market', `从NPC商店购买了${item.name}，花费${price}金币`);
   464→   464→  }, [character, inventory, addLog]);
   465→   465→
   466→   466→  // 结束战斗
   467→   467→  const endBattle = useCallback(() => {
   468→   468→    setBattle({
   469→   469→      inBattle: false,
   470→   470→      monster: null,
   471→   471→      playerHp: 0,
   472→   472→      monsterHp: 0,
   473→   473→      battleLog: [],
   474→   474→      isAuto: false,
